\[

\[
y'' = 2 + \frac{8}{(x-1)^3} \qquad - + \frac{4(-1)^n}{(x-1)^{n+1}} x^n

\[
y''' = \frac{-24}{(x-1)^4}

\[
y^{(n)} = \frac{(-1)^n \cdot n!}{(x-1)^{n+1}}

\[
(2)\ y' = 2\sin x \cos x = \sin 2x

\[
y'' = 2 \cdot \sin(2x + \frac{\pi}{2})
\]

\[
y''' = 2^2 \sin(2x + \pi)
\]

\[
\therefore y = 0 + 0 + x + \cdots - \frac{2^n \sin(2x + n\frac{\pi}{2})}{n!} \cdot x^n
