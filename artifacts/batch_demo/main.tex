3.6.1. (1)

\[
y = x^2 + x + 3 + \frac{4}{x-1}
\]

\[
y' = 2x + 1 + \frac{-4}{(x-1)^2}
\]

\[
y'' = 2 + \frac{8}{(x-1)^3}
\]

\[
y''' = \frac{-24}{(x-1)^4}
\]

\[
y^{(n)} = \frac{4(-1)^n \cdot n!}{(x-1)^{n+1}}
\]

\[
\therefore y = -1 - 3x + 5x^2 - 4x^3 + \cdots + \frac{4(-1)^n}{(\xi-1)^{n+1}}x^n
\]

(2)

\[
y' = 2\sin x\cos x = \sin 2x
\]

\[
y'' = 2 \cdot \sin\left(2x + \frac{\pi}{2}\right)
\]

\[
y''' = 2^2 \sin(2x + \pi)
\]

\[
\therefore y = 0 + 0 + x + \cdots - \frac{2^n \sin\left(2x + n\frac{\pi}{2}\right)}{n!}x^n
\]

\[
3.6.6\ (1)\quad \text{原式}=\lim_{x\to0}\frac{\left[1-\frac{1}{2}x^{2}+\frac{1}{24}x^{4}-\left(1+\left(-\frac{1}{2}x^{2}\right)+\frac{1}{2}\left(-\frac{1}{2}x^{2}\right)^{2}+o(x^{4})\right)\right]}{x^{4}}

\[
=\lim_{x\to0}\frac{\frac{1}{24}x^{4}-\frac{1}{8}x^{4}}{x^{4}}
\]

\[

\]

\[
(2)\quad \lim_{x\to0}\frac{1+\frac{1}{6}x^{2}+o(x^{2})-\left(1-\frac{1}{6}x^{2}+o(x^{2})\right)}{x^{2}}

\[
=\frac{1}{3}
\]

\[
(3)\quad \text{原式}=\lim_{t\to0}\frac{t-\ln(1+t)}{t^{2}}
\]

\[
=\lim_{t\to0}\frac{t-\left(t-\frac{1}{2}t^{2}\right)}{t^{2}}
\]

\[
=\frac{1}{2}
\]

\[
(4)\quad \lim_{x\to0}\frac{1-\frac{1}{2}\sin^{2}x+\frac{1}{24}\sin^{4}x-\left(1-\frac{1}{2}x^{2}+\frac{1}{24}x^{4}\right)}{x^{4}}

\[
=\lim_{x\to0}\frac{1-\frac{1}{2}\left(x-\frac{1}{6}x^{3}\right)^{2}+\frac{1}{24}\left(x-\frac{1}{6}x^{3}\right)^{4}-\left(1-\frac{1}{2}x^{2}+\frac{1}{24}x^{4}\right)+o(x^{4})}{x^{4}}

\[
