\[
x^3 - x^2 + x^2 + x - x + 3x - 3 + 4
\]

3.6.1. (1)

\[
y = x^2 + x + 3 + \frac{4}{x-1}
\]

\[
y' = 2x + 1 + \frac{-4}{(x-1)^2}
\]

\[
y'' = 2 + \frac{8}{(x-1)^3}
\]

\[
y''' = \frac{-24}{(x-1)^4}
\]

\[
y^{(n)} = \frac{4(-1)^n \cdot n!}{(x-1)^{n+1}}
\]

\[
\therefore y = -1 - 3x + 5x^2 - 4x^3 + \cdots + \frac{4(-1)^n}{(\xi-1)^{n+1}}x^n
\]

(2)

\[
y' = 2\sin x\cos x = \sin 2x
\]

\[
y'' = 2 \cdot \sin\left(2x + \frac{\pi}{2}\right)
\]

\[
y''' = 2^2 \sin(2x + \pi)
\]

\[
\therefore y = 0 + 0 + x + \cdots - \frac{2^n \sin\left(2x + n\frac{\pi}{2}\right)}{n!}x^n
\]
