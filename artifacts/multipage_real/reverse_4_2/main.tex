\[
\because\quad f(x_1)=f(x_0)+f'(x_0)(x_1-x_0)+\frac{f''(\xi_1)}{2!}(x_1-x_0)^2
\]

\[
f(x_2)=f(x_0)+f'(x_0)(x_2-x_0)+\frac{f''(\xi_2)}{2!}(x_2-x_0)^2
\]

令 \textbackslash{}(x\_1=0,\textbackslash{} x\_2=2\textbackslash{})

\[
f(0)=f(x_0)+f'(x_0)(-x_0)+\frac{f''(\xi_1)}{2!}(-x_0)^2
\]

\[
f(2)=f(x_0)+f'(x_0)(2-x_0)+\frac{f''(\xi_2)}{2!}(2-x_0)^2
\]

\[
\Rightarrow\quad 2|f(x_0)|\le |f(0)|+|f(2)|+\cdots
\]

\[
\le 4
\]

\[
\Rightarrow\quad |f(x_0)|\le 2
\]
